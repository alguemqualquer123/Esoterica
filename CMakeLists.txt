#-------------------------------------------------------------------------
# Esoterica — CMake build (migration from MSBuild)
#-------------------------------------------------------------------------
# This file mirrors the official MSBuild setup (Esoterica.slnx +
# Code/PropertySheets/Esoterica.props) so that the same code can be built
# with CMake. It is a starting point for the migration:
#
#   * Same project structure and dependency graph as the .slnx
#   * Same configurations: Debug / Release / Shipping
#   * Same compile settings: C++20, exceptions disabled, /Wall + /WX,
#     /MD, NDEBUG on all x64 configs, EE_DEBUG / EE_RELEASE / EE_SHIPPING
#   * Modules are built STATIC by default (like the MSBuild Shipping config)
#     — pass -DEE_MODULES_SHARED=ON to get DLL builds like Debug/Release
#
# Known limitations / TODO:
#   * The Reflector must be run BEFORE the first build to generate the
#     type-info / shader code in <module>/_Module/_AutoGenerated
#     (see Code/Scripts/Reflect/Reflect.nmake: EsotericaReflector -s Esoterica.slnx)
#   * External SDKs are expected under ./External with the same layout as
#     the MSBuild property sheets (DXC, Freetype, GameNetworkingSockets,
#     ixWebSocket, RenderDoc, AMD AGS, WinPix, DbgHelp, ...). Every SDK
#     reference is conditional — configure succeeds without them, compile
#     of the code that needs them will not.
#   * Shader compilation (.esh/.esf -> DXIL) is not wired up yet; the
#     compiled shader bytecode is embedded via the Reflector/ResourceCompiler
#     pipeline in the MSBuild flow.
#-------------------------------------------------------------------------

cmake_minimum_required( VERSION 3.24 )

project( Esoterica
    VERSION     0.1.0
    DESCRIPTION "Esoterica Game Engine"
    LANGUAGES   C CXX
)

#-------------------------------------------------------------------------
# Platform checks
#-------------------------------------------------------------------------

# Supported platforms: Windows x64 (MSVC/clang-cl) and Linux x64 (GCC/Clang)
if( WIN32 )
    set( EE_PLATFORM_WINDOWS ON )
    set( EE_PLATFORM_LINUX OFF )
elseif( UNIX AND NOT APPLE )
    set( EE_PLATFORM_LINUX ON )
    set( EE_PLATFORM_WINDOWS OFF )
    # Linux requires Vulkan instead of D3D12
    message( STATUS "Esoterica: Linux platform detected — Vulkan backend will be used" )
else()
    message( FATAL_ERROR "Unsupported platform — only Windows x64 and Linux x64 are supported (macOS is not yet supported)" )
endif()

if( EE_PLATFORM_WINDOWS AND NOT MSVC AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" )
    message( FATAL_ERROR "On Windows only MSVC and Clang (clang-cl) are supported" )
endif()

if( EE_PLATFORM_LINUX AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" )
    message( WARNING "Linux builds are tested with GCC and Clang — compiler '${CMAKE_CXX_COMPILER_ID}' may not be fully supported" )
endif()

if( NOT CMAKE_SIZEOF_VOID_P EQUAL 8 )
    message( FATAL_ERROR "Esoterica only supports 64-bit builds (x64)" )
endif()

#-------------------------------------------------------------------------
# Options
#-------------------------------------------------------------------------

# Dev tools (EngineTools, GameTools, Editor, ResourceCompiler, ResourceServer,
# Reflector, Tester, BuildGenerator) are disabled for Shipping, mirroring
# <Build Solution="Shipping|*" Project="false" /> in the .slnx
if( CMAKE_BUILD_TYPE STREQUAL "Shipping" )
    set( EE_ENABLE_DEV_TOOLS_DEFAULT OFF )
else()
    set( EE_ENABLE_DEV_TOOLS_DEFAULT ON )

option( EE_ENABLE_DEV_TOOLS "Build development tools (EngineTools, GameTools, Editor, etc.)" ${EE_ENABLE_DEV_TOOLS_DEFAULT} )

# Mirror Debug/Release module DLL builds. OFF == static libraries (Shipping semantics).
option( EE_MODULES_SHARED "Build engine modules as DLLs (EE_DLL), like the MSBuild Debug/Release configs" OFF )

# Esoterica.props sets TreatWarningAsError=true — keep that by default.
option( EE_WARNINGS_AS_ERRORS "Treat warnings as errors (/WX)" ON )

# Configurations
#-------------------------------------------------------------------------

set( EE_IS_SHIPPING FALSE )
if( CMAKE_BUILD_TYPE STREQUAL "Shipping" )
    set( EE_IS_SHIPPING TRUE )
endif()

# Language standards — Esoterica.props: stdcpp20 + stdc17
#-------------------------------------------------------------------------

set( CMAKE_CXX_STANDARD 20 )
set( CMAKE_CXX_STANDARD_REQUIRED ON )
set( CMAKE_CXX_EXTENSIONS OFF )

set( CMAKE_C_STANDARD 17 )
set( CMAKE_C_STANDARD_REQUIRED ON )
set( CMAKE_C_EXTENSIONS OFF )

# Esoterica.props: <RuntimeLibrary>MultiThreadedDLL</RuntimeLibrary> for ALL
# configurations (Debug included) — release CRT, never the debug CRT.
# Linux: no MSVC runtime concept — use default.
if( MSVC )
    set( CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL" )
endif()

#-------------------------------------------------------------------------
# Common compile definitions
#-------------------------------------------------------------------------
# From Esoterica.props:
#   NDEBUG;NOMINMAX;WIN32_LEAN_AND_MEAN;_CRT_SECURE_NO_WARNINGS  (x64)
#   _HAS_EXCEPTIONS=0
#   EE_DEBUG=1;EE_DLL   (Debug)   /   EE_RELEASE=1;EE_DLL   (Release)
#   EE_SHIPPING=1       (Shipping)
#
# NOTE: NDEBUG is intentionally defined for Debug as well — Esoterica.props
# defines it for every x64 configuration and the code base relies on it.

set( EE_EASTL_USER_CONFIG_HEADER "${CMAKE_SOURCE_DIR}/Code/Base/ThirdParty/EA/eastl_Esoterica.h" )

if( WIN32 )
    add_compile_definitions(
        NOMINMAX
        WIN32_LEAN_AND_MEAN
        _CRT_SECURE_NO_WARNINGS
    )
endif()

add_compile_definitions(
    _HAS_EXCEPTIONS=0
    "EASTL_USER_CONFIG_HEADER=\"${EE_EASTL_USER_CONFIG_HEADER}\""
    $<$<CONFIG:Debug>:EE_DEBUG=1>
    $<$<CONFIG:Debug>:NDEBUG>
    $<$<CONFIG:Release>:EE_RELEASE=1>
    $<$<CONFIG:Shipping>:EE_SHIPPING=1>
    $<$<CONFIG:Shipping>:NDEBUG>
)

if( EE_PLATFORM_LINUX )
    add_compile_definitions( EE_PLATFORM_LINUX=1 )
elseif( EE_PLATFORM_WINDOWS )
    add_compile_definitions( EE_PLATFORM_WINDOWS=1 )
endif()

if( EE_MODULES_SHARED )
    # EA.props defines EASTL_DLL for everything except Shipping
    add_compile_definitions( EE_DLL EASTL_DLL )
endif()

#-------------------------------------------------------------------------
# Common compile options
#-------------------------------------------------------------------------

# Warning policy — Esoterica.props: WarningLevel=EnableAllWarnings (/Wall)
# plus an explicit list of disabled warnings, and TreatWarningAsError.
if( MSVC )
    set( EE_DISABLED_WARNINGS
        /wd4865 /wd4189 /wd4946 /wd4191 /wd5220 /wd4255 /wd4266 /wd4623
        /wd4711 /wd4388 /wd4355 /wd5031 /wd4774 /wd4582 /wd4365 /wd5246
        /wd4061 /wd5219 /wd4514 /wd4505 /wd4251 /wd4100 /wd4127 /wd4201
        /wd4577 /wd4464 /wd4668 /wd4710 /wd4820 /wd5052 /wd4619 /wd4625
        /wd4626 /wd5026 /wd5027 /wd5045 /wd26495 /wd5262 /wd5264 /wd5267
    )

    add_compile_options(
        /MP              # MultiProcessorCompilation
        /bigobj
        /Zo
        /Zc:inline
        /EHs-c-          # ExceptionHandling=false
        /fp:precise      # FloatingPointModel=Precise
        /Wall
        ${EE_DISABLED_WARNINGS}
    )

    if( EE_WARNINGS_AS_ERRORS )
        add_compile_options( /WX )
    endif()

    # Optimization / codegen per configuration (Esoterica.props)
    add_compile_options(
        $<$<CONFIG:Debug>:/Od>      # Optimization=Disabled
        $<$<CONFIG:Debug>:/Ob1>     # InlineFunctionExpansion=OnlyExplicitInline
        $<$<CONFIG:Debug>:/Oi>      # IntrinsicFunctions=true
        $<$<CONFIG:Debug>:/Gy>      # FunctionLevelLinking=true
        $<$<CONFIG:Release>:/O2>    # Optimization=MaxSpeed
        $<$<CONFIG:Release>:/Ob2>   # InlineFunctionExpansion=AnySuitable
        $<$<CONFIG:Release>:/Gy>
        $<$<CONFIG:Shipping>:/O2>
        $<$<CONFIG:Shipping>:/Ob2>
        $<$<CONFIG:Shipping>:/Gy>
        $<$<CONFIG:Shipping>:/GL>   # WholeProgramOptimization=true (Shipping only)
    )

    # Debug info — GenerateDebugInformation=DebugFull for everything except Shipping
    add_compile_options( $<$<NOT:$<CONFIG:Shipping>>:/Zi> )
    add_link_options( $<$<NOT:$<CONFIG:Shipping>>:/DEBUG> )
    add_link_options( $<$<CONFIG:Shipping>:/LTCG> )
else()
    # GCC / Clang (Linux) — mirror MSVC intent with portable flags
    add_compile_options(
        -Wall -Wextra -Wpedantic
        -Wno-unused-parameter -Wno-unused-variable -Wno-missing-field-initializers
        -Wno-unknown-pragmas -Wno-sign-compare -Wno-type-limits
        -fPIC
        $<$<CONFIG:Debug>:-O0>
        $<$<CONFIG:Debug>:-g>
        $<$<CONFIG:Release>:-O2>
        $<$<CONFIG:Release>:-g>
        $<$<CONFIG:Shipping>:-O2>
        $<$<CONFIG:Shipping>:-flto>
    )
    add_link_options( $<$<CONFIG:Shipping>:-flto> )
    if( EE_WARNINGS_AS_ERRORS )
        add_compile_options( -Werror )
    endif()
endif()

#-------------------------------------------------------------------------
# External SDKs (see Code/PropertySheets/*.props)
#-------------------------------------------------------------------------
# All SDK references are conditional: they only take effect when the SDK
# directory exists. Configure always succeeds; compiling code that needs a
# missing SDK will fail, exactly like the MSBuild property sheets would.
#
# TODO: Wire up the remaining optional SDKs used by dev tools:
#   LivePP, Optick, SuperLuminal, Navpower, FBXSDK, SQLite, Box3D,
#   MeshOptimizer, CTT, LLVM, RenderDoc (runtime), AmdAgs, WinPixEventRuntime

set( EE_EXTERNAL_DIR "${CMAKE_SOURCE_DIR}/External" CACHE PATH "Directory with third-party SDKs (same layout as the MSBuild property sheets)" )

# Platform-specific system libs
if( EE_PLATFORM_WINDOWS )
    set( EE_LINK_LIBS dxguid d3d12 dxgi Xinput )   # Esoterica.Win32.props
elseif( EE_PLATFORM_LINUX )
    # Vulkan replaces D3D12; X11/Wayland/SDL handled via find_package or pkg-config
    set( EE_LINK_LIBS pthread dl m )
    # Vulkan loader — try system package first, fallback to External/Vulkan
    find_package( Vulkan QUIET )
    if( Vulkan_FOUND )
        list( APPEND EE_LINK_LIBS Vulkan::Vulkan )
    else()
        # Fallback: expect External/Vulkan or system libvulkan.so
        list( APPEND EE_LINK_LIBS vulkan )
    endif()
    # SDL3 for windowing/input on Linux (optional — falls back to X11)
    find_package( SDL3 QUIET )
    if( SDL3_FOUND )
        list( APPEND EE_LINK_LIBS SDL3::SDL3 )
        add_compile_definitions( EE_WITH_SDL3=1 )
    else()
        # Try pkg-config for SDL3 or SDL2
        find_package( PkgConfig QUIET )
        if( PkgConfig_FOUND )
            pkg_check_modules( SDL3 QUIET sdl3 )
            if( SDL3_FOUND )
                include_directories( ${SDL3_INCLUDE_DIRS} )
                list( APPEND EE_LINK_LIBS ${SDL3_LIBRARIES} )
                add_compile_definitions( EE_WITH_SDL3=1 )
            else()
                pkg_check_modules( SDL2 QUIET sdl2 )
                if( SDL2_FOUND )
                    include_directories( ${SDL2_INCLUDE_DIRS} )
                    list( APPEND EE_LINK_LIBS ${SDL2_LIBRARIES} )
                    add_compile_definitions( EE_WITH_SDL2=1 )
                endif()
            endif()
        endif()
    endif()
endif()
set( EE_RUNTIME_DLLS "" )

# DXC (DirectXShaderCompiler) — DXC.props
set( EE_DXC_DIR "${EE_EXTERNAL_DIR}/DirectXShaderCompiler" CACHE PATH "DXC SDK directory" )
if( EXISTS "${EE_DXC_DIR}/inc" )
    include_directories( "${EE_DXC_DIR}/inc" )
    if( WIN32 )
        link_directories( "${EE_DXC_DIR}/lib/x64" )
        list( APPEND EE_LINK_LIBS dxcompiler )
        if( EXISTS "${EE_DXC_DIR}/bin/x64/dxcompiler.dll" )
            list( APPEND EE_RUNTIME_DLLS "${EE_DXC_DIR}/bin/x64/dxcompiler.dll" )
        endif()
        if( EXISTS "${EE_DXC_DIR}/bin/x64/dxil.dll" )
            list( APPEND EE_RUNTIME_DLLS "${EE_DXC_DIR}/bin/x64/dxil.dll" )
        endif()
    else()
        # Linux: DXC ships as libdxcompiler.so in lib/x64 or lib/
        if( EXISTS "${EE_DXC_DIR}/lib/x64/libdxcompiler.so" )
            link_directories( "${EE_DXC_DIR}/lib/x64" )
            list( APPEND EE_LINK_LIBS dxcompiler )
        elseif( EXISTS "${EE_DXC_DIR}/lib/libdxcompiler.so" )
            link_directories( "${EE_DXC_DIR}/lib" )
            list( APPEND EE_LINK_LIBS dxcompiler )
        else()
            # Try system dxc
            find_library( DXC_LIB dxcompiler )
            if( DXC_LIB )
                list( APPEND EE_LINK_LIBS dxcompiler )
            endif()
        endif()
    endif()
endif()

# FreeType — FreeType.props (cross-platform — also try pkg-config on Linux)

    include_directories( "${EE_FREETYPE_DIR}/include" )

    else()
        if( EXISTS "${EE_FREETYPE_DIR}/lib/libfreetype.so" OR EXISTS "${EE_FREETYPE_DIR}/build/libfreetype.so" )
            link_directories( "${EE_FREETYPE_DIR}/lib" "${EE_FREETYPE_DIR}/build" )

        endif()
    endif()
endif()

#-------------------------------------------------------------------------
# Helpers
#-------------------------------------------------------------------------

# Collects the sources of a module. Third-party sources compiled inside the
# module (matching the .vcxproj) are included; only known non-buildable
# trees are excluded.
# On Linux, Win32-specific source files are excluded automatically to avoid
# missing header errors — the Linux counterparts are included instead.
function( ee_collect_sources OUT_VAR ROOT_DIR )
    file( GLOB_RECURSE _sources CONFIGURE_DEPENDS
        "${ROOT_DIR}/*.cpp"
        "${ROOT_DIR}/*.cxx"
        "${ROOT_DIR}/*.cc"
        "${ROOT_DIR}/*.c"
        "${ROOT_DIR}/*.h"
        "${ROOT_DIR}/*.hpp"
    )

    # Generated code is added separately (only when the Reflector has run)
    list( FILTER _sources EXCLUDE REGEX "[\\\\/]_AutoGenerated[\\\\/]" )
    list( FILTER _sources EXCLUDE REGEX "[\\\\/]_Autogenerated[\\\\/]" )

    # Third-party trees that are not part of the module build
    list( FILTER _sources EXCLUDE REGEX "ThirdParty[\\/]pugixml[\\/]docs" )
    list( FILTER _sources EXCLUDE REGEX "ThirdParty[\\/]rapidhash[\\/]collisions" )
    list( FILTER _sources EXCLUDE REGEX "ThirdParty[\\/]rapidhash[\\/]bench" )
    list( FILTER _sources EXCLUDE REGEX "ThirdParty[\\\\/]ufbx[\\\\/](examples|bindgen|extra|misc|test)" )
    list( FILTER _sources EXCLUDE REGEX "ThirdParty[\\\\/]box3d[\\\\/](samples|docs|shared)" )

    # Platform-specific filtering
    if( EE_PLATFORM_LINUX )
        # Exclude Win32 sources on Linux — Linux counterparts will be used
        list( FILTER _sources EXCLUDE REGEX "_Win32\\.cpp$" )
        list( FILTER _sources EXCLUDE REGEX "_Win32\\.h$" )
        list( FILTER _sources EXCLUDE REGEX "Platform/Win32" )
        list( FILTER _sources EXCLUDE REGEX "RHI_Direct3D12\\.cpp$" )
        list( FILTER _sources EXCLUDE REGEX "D3D12MemAlloc\\.cpp$" )
        # WinPix / AGS are Windows-only
        list( FILTER _sources EXCLUDE REGEX "WinPix" )
        list( FILTER _sources EXCLUDE REGEX "amd_ags" )
    elseif( EE_PLATFORM_WINDOWS )
        # Exclude Linux sources on Windows
        list( FILTER _sources EXCLUDE REGEX "_Linux\\.cpp$" )
        list( FILTER _sources EXCLUDE REGEX "_Linux\\.h$" )
        list( FILTER _sources EXCLUDE REGEX "Platform/Linux" )
        list( FILTER _sources EXCLUDE REGEX "RHI_Vulkan\\.cpp$" )
    endif()

    set( ${OUT_VAR} ${_sources} PARENT_SCOPE )
endfunction()

# Appends Reflector-generated sources if they exist
# (Code/Scripts/Reflect/Reflect.nmake generates them into _Module/_AutoGenerated)
function( ee_append_autogenerated OUT_VAR ROOT_DIR )
    # Both spellings exist in the code base (_AutoGenerated / _Autogenerated)
    # but refer to the same directory on a case-insensitive filesystem, so
    # only glob the first one that exists to avoid adding every file twice.
    foreach( gen_dir "${ROOT_DIR}/_Module/_AutoGenerated" "${ROOT_DIR}/_Module/_Autogenerated" )
        if( EXISTS "${gen_dir}" )
            file( GLOB_RECURSE _gen_sources CONFIGURE_DEPENDS "${gen_dir}/*.cpp" "${gen_dir}/*.h" )
            list( APPEND ${OUT_VAR} ${_gen_sources} )
            break()
        endif()
    endforeach()
    set( ${OUT_VAR} ${${OUT_VAR}} PARENT_SCOPE )
endfunction()

# Applies per-target settings common to all modules/applications
function( ee_configure_target TARGET PROJECT_DEFINE )
    # ESOTERICA_<PROJECTNAME> is required by Code/<module>/_Module/API.h for
    # dllexport/dllimport when EE_DLL is defined (mirrors
    # $(ProjectName.ToUpper().Replace('.', '_')) in Esoterica.props)
    target_compile_definitions( ${TARGET} PRIVATE ${PROJECT_DEFINE} )

    target_link_libraries( ${TARGET} PRIVATE ${EE_LINK_LIBS} )

    set_target_properties( ${TARGET} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${EE_OUTPUT_DIR}"
        LIBRARY_OUTPUT_DIRECTORY "${EE_OUTPUT_DIR}"
        ARCHIVE_OUTPUT_DIRECTORY "${EE_OUTPUT_DIR}"
        PDB_OUTPUT_DIRECTORY    "${EE_OUTPUT_DIR}"
    )

    # Copy external SDK runtime DLLs next to the produced binaries
    foreach( dll IN LISTS EE_RUNTIME_DLLS )
        add_custom_command( TARGET ${TARGET} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:${TARGET}>"
        )
    endforeach()
endfunction()

#-------------------------------------------------------------------------
# Modules
#-------------------------------------------------------------------------

function( ee_add_module NAME PROJECT_DEFINE REL_DIR )
    set( _root "${CMAKE_SOURCE_DIR}/${REL_DIR}" )
    ee_collect_sources( _sources "${_root}" )
    ee_append_autogenerated( _sources "${_root}" )

    if( EE_MODULES_SHARED )
        add_library( ${NAME} SHARED ${_sources} )
    else()
        add_library( ${NAME} STATIC ${_sources} )
    endif()

    ee_configure_target( ${NAME} ${PROJECT_DEFINE} )
    set_target_properties( ${NAME} PROPERTIES FOLDER "Modules" )
endfunction()

#-------------------------------------------------------------------------
# Applications
#-------------------------------------------------------------------------

function( ee_add_application NAME OUTPUT_NAME PROJECT_DEFINE REL_DIR )
    set( _root "${CMAKE_SOURCE_DIR}/${REL_DIR}" )
    ee_collect_sources( _sources "${_root}" )
    ee_append_autogenerated( _sources "${_root}" )

    add_executable( ${NAME} ${_sources} )
    set_target_properties( ${NAME} PROPERTIES
        OUTPUT_NAME "${OUTPUT_NAME}"
        FOLDER      "Applications"
    )
    ee_configure_target( ${NAME} ${PROJECT_DEFINE} )
endfunction()
